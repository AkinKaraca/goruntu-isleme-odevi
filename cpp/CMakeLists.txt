cmake_minimum_required(VERSION 3.18)
project(goruntu_boyutlandir LANGUAGES CXX)
option(ENABLE_CUDA "OpenCV CUDA resize destegi" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
if(ENABLE_CUDA)
    find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs cudawarping)
else()
    find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs)
endif()
add_executable(boyutlandir resize_images.cpp)
target_include_directories(boyutlandir PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(boyutlandir PRIVATE ${OpenCV_LIBS} Threads::Threads)
if(ENABLE_CUDA)
    target_compile_definitions(boyutlandir PRIVATE ENABLE_CUDA=1)
    add_executable(cuda_kontrol cuda_check.cpp)
    target_include_directories(cuda_kontrol PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_link_libraries(cuda_kontrol PRIVATE ${OpenCV_LIBS})
endif()
if(MSVC)
    target_compile_options(boyutlandir PRIVATE /W4 /utf-8)
else()
    target_compile_options(boyutlandir PRIVATE -Wall -Wextra -Wpedantic)
endif()
